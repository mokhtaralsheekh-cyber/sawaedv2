        function calculateShipment(s) {

            let totalSpaceUnits = 0;

            s.items.forEach(item => {

                totalSpaceUnits += (Number(item.qty) || 0) * (Number(item.spaceFactor) || 1);

            });

            const totalSeaFreightLYD = (Number(s.seaFreightUSD) || 0) * (Number(s.exchangeRate) || 0);

            const totalCustomsLYD = (Number(s.customsUSD) || 0) * (Number(s.exchangeRate) || 0);

            const totalGeneralExpensesLYD = totalSeaFreightLYD + totalCustomsLYD + (Number(s.localTransportLYD) || 0) + (Number(s.otherExpensesLYD) || 0);

            const itemsCalculated = s.items.map(item => {

                const qty = Number(item.qty) || 0;

                const fobUSD = Number(item.fobUSD) || 0;

                const spaceFactor = Number(item.spaceFactor) || 1;

               

                const fobLYD = fobUSD * (Number(s.exchangeRate) || 0);

                const itemTotalSpace = qty * spaceFactor;

                const shareOfExpensesLYD = totalSpaceUnits > 0 ? (itemTotalSpace / totalSpaceUnits) * totalGeneralExpensesLYD : 0;

                const totalCostLYD = (qty * fobLYD) + shareOfExpensesLYD;

                const unitCostLYD = qty > 0 ? totalCostLYD / qty : 0;

                const unitCostUSD = s.exchangeRate > 0 ? unitCostLYD / s.exchangeRate : 0;

                const sellingPriceLYD = unitCostLYD * (1 + ((Number(s.profitMargin) || 0) / 100));

                const totalSellingLYD = sellingPriceLYD * qty;

                return { ...item, fobLYD, unitCostUSD, unitCostLYD, sellingPriceLYD, totalSellingLYD };

            });

            return { totalSpaceUnits, totalGeneralExpensesLYD, itemsCalculated };

        }

        async function handleAIInvoiceUpload(event) {

            const file = event.target.files[0];

            if (!file) return;

            const apiKey = appData.apiKey;

            if (!apiKey) {

                alert("يرجى إدخال مفتاح Gemini API في الإعدادات أولاً.");

                return;

            }

            const banner = document.getElementById("aiLoadingBanner");

            banner.classList.remove("hidden");

            try {

                const base64Data = await new Promise((resolve, reject) => {

                    const reader = new FileReader();

                    reader.onload = () => resolve(reader.result.split(',')[1]);

                    reader.onerror = reject;

                    reader.readAsDataURL(file);

                });

                const promptText = `Analyze this commercial invoice image for tires or batteries. Extract all line items into a strict JSON array. Each item must have:

- "brand": string (e.g., DEWOSTONE, PHLORIDE, LINGLONG, AOTELI, FULLRIDE, etc.)

- "size": string (tire size like 195/65R15 or battery capacity like 150Ah)

- "type": string ("PCR", "LTR", "TBR", or "BATTERY")

- "qty": number (quantity)

- "fobUSD": number (unit price in USD)

- "spaceFactor": number (use 2 for TBR truck tires or >=150Ah batteries, 1.5 for LTR/C tires, 1 for standard PCR tires or <=100Ah batteries)

Return ONLY valid raw JSON array without markdown formatting.`;

                const models = ["gemini-2.5-flash", "gemini-2.0-flash", "gemini-1.5-flash"];

                let parsedItems = null;

                let lastError = "";

                for (const model of models) {

                    try {

                        const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${encodeURIComponent(apiKey)}`;

                        const res = await fetch(url, {

                            method: 'POST',

                            headers: { 'Content-Type': 'application/json' },

                            body: JSON.stringify({

                                contents: [{

                                    parts: [

                                        { text: promptText },

                                        { inline_data: { mime_type: file.type || "image/jpeg", data: base64Data } }

                                    ]

                                }]

                            })

                        });

                        const data = await res.json();

                        if (data.error) {

                            lastError = data.error.message;

                            continue;

                        }

                        const textResult = data.candidates[0].content.parts[0].text;

                        const cleanJson = textResult.replace(/```json/gi, '').replace(/```/g, '').trim();

                        parsedItems = JSON.parse(cleanJson);

                        if (Array.isArray(parsedItems) && parsedItems.length > 0) break;

                    } catch (e) {

                        lastError = e.message;

                    }

                }

                if (!parsedItems || !Array.isArray(parsedItems) || parsedItems.length === 0) {

                    throw new Error(lastError || "لم يتم العثور على أصناف واضحة في الصورة.");

                }

                const newShipId = "ship_" + Date.now();

                appData.shipments.push({

                    id: newShipId,

                    name: "فاتورة AI (" + new Date().toLocaleDateString('ar-LY') + ")",

                    exchangeRate: getActiveShipment().exchangeRate || 4.85,

                    customsUSD: 1000,

                    seaFreightUSD: 4000,

                    localTransportLYD: 1000,

                    otherExpensesLYD: 300,

                    profitMargin: 15,

                    items: parsedItems.map((pi, idx) => ({

                        id: Date.now() + idx,

                        code: "AI-" + (idx + 1),

                        brand: (pi.brand || "DEWOSTONE").toUpperCase(),

                        size: pi.size || "Standard",

                        type: pi.type || "PCR",

                        qty: Number(pi.qty) || 100,

                        fobUSD: Number(pi.fobUSD) || 20,

                        spaceFactor: Number(pi.spaceFactor) || 1

                    }))

                });

                saveStorage();

                switchShipment(newShipId);

                alert("✅ تم استخراج " + parsedItems.length + " صنف من الفاتورة وإنشاء شحنة جديدة بنجاح!");

            } catch (err) {

                console.error(err);

                alert("❌ تعذر قراءة الفاتورة: " + err.message + "\nتأكد من صحة مفتاح Gemini API في الإعدادات.");

            } finally {

                banner.classList.add("hidden");

            }

        }
